🔧 One-time Setup

This scanner reads coupon data from your Google Sheet and writes the QR Status column (TRUE/FALSE) back when a coupon is redeemed. You need to deploy a Google Apps Script Web App once.

Your sheet columns detected: Timestamp, Email Address, NAME, ROLL NO, EMAIL ID, utr, Phone Number, OPTION, AMOUNT PAID, QR Status, tot

Step 1 — Open Apps Script

In your Google Sheet, go to Extensions → Apps Script. Delete any existing code and paste this:

const SHEET_ID = '1NS0yAI7jxcFg3xai2fOiwveGu1szPNj4gkotXLUaeKk'; const SHEET_NAME = 'Sheet1'; function doGet(e) { if (e.parameter.action === 'getData') return getData(); return ContentService.createTextOutput('Unknown action').setMimeType(ContentService.MimeType.TEXT); } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); if (payload.action === 'updateStatus') return updateStatus(payload.id, payload.status, payload.chosenFood, payload.usedAt); return jsonRes({ error: 'Unknown action' }); } catch(err) { return jsonRes({ error: err.toString() }); } } function getData() { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const data = sheet.getDataRange().getValues(); const headers = data[0].map(h => h.toString().trim().toLowerCase()); const rows = []; for (let i = 1; i < data.length; i++) { const row = {}; headers.forEach((h, j) => row[h] = data[i][j]); rows.push(row); } return ContentService.createTextOutput(JSON.stringify({ rows, headers })) .setMimeType(ContentService.MimeType.JSON); } function updateStatus(id, status, chosenFood, usedAt) { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const data = sheet.getDataRange().getValues(); const headers = data[0].map(h => h.toString().trim().toLowerCase()); const idCol = headers.indexOf('utr'); // PRIMARY KEY const rollCol = headers.indexOf('roll no'); // fallback const statusCol = headers.indexOf('qr status'); // column to update const foodCol = headers.indexOf('option'); // food choice column const emailCol = headers.indexOf('email id'); // for extra context // Optional log columns — create if missing let chosenFoodCol = headers.indexOf('chosenfood'); let usedAtCol = headers.indexOf('usedat'); if (chosenFoodCol === -1) { chosenFoodCol = headers.length; sheet.getRange(1, chosenFoodCol + 1).setValue('chosenfood'); } if (usedAtCol === -1) { usedAtCol = headers.length + (chosenFoodCol === headers.length ? 1 : 0) + 1; sheet.getRange(1, usedAtCol + 1).setValue('usedat'); } if (statusCol === -1) return jsonRes({ error: 'QR Status column not found in sheet' }); if (idCol === -1 && rollCol === -1) return jsonRes({ error: 'Neither utr nor roll no column found' }); const lookupCol = idCol !== -1 ? idCol : rollCol; for (let i = 1; i < data.length; i++) { if (data[i][lookupCol].toString().trim() === id.toString().trim()) { sheet.getRange(i + 1, statusCol + 1).setValue(status); if (chosenFood && foodCol !== -1) sheet.getRange(i + 1, foodCol + 1).setValue(chosenFood); if (chosenFood && chosenFoodCol !== -1) sheet.getRange(i + 1, chosenFoodCol + 1).setValue(chosenFood); sheet.getRange(i + 1, usedAtCol + 1).setValue(usedAt || new Date()); return jsonRes({ success: true, row: i + 1, id: id }); } } return jsonRes({ error: 'ID not found: ' + id }); } function jsonRes(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }

💡 Column mapping for your sheet:
Primary key: utr (falls back to roll no)
Status written to: qr status
Food option read from: option
